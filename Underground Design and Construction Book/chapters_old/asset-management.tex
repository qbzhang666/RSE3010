\chapter{Operation, Inspection and Asset Management}
\label{ch:asset-management}


\section{Transition to operation and deterioration}
\label{sec:transition-to-operation-and-deterioration}

\subsection{From construction to operation}
\label{sec:handover}
\scopenote{Handover information, as-built records, residual risk registers and
the baseline condition survey. The information the operator needs and rarely
receives in usable form.}

\subsection{Deterioration mechanisms}
\label{sec:deterioration}
\scopenote{Water ingress and leaching, reinforcement corrosion, freeze-thaw,
chemical attack, fatigue from dynamic loading, and slow ground movement
loading the lining.}

\section{Inspection and condition assessment}
\label{sec:inspection-and-condition-assessment}

\subsection{Inspection regimes and condition rating}
\label{sec:inspection}
\scopenote{Principal and general inspection intervals, access and possession
constraints, defect classification and condition rating scales that support
prioritisation.}

\subsection{Inspection technologies}
\label{sec:inspection-tech}
\scopenote{Mobile laser scanning and imaging trains, ground penetrating radar
for lining thickness and voids, impact echo, thermography, and automated
defect detection.}

\subsection{From inspection data to predictive maintenance}
\label{sec:predictive-maintenance}
\scopenote{Surface observation, internal inspection by ground penetrating
radar and scanning, and multi-physics modelling combine to forecast condition
rather than record it. This is what moves asset management from reactive
repair to forecast-based intervention, and it depends on inspection data
returning to the model of Chapter~\ref{ch:digital-underground}.}

\section{Assessment of structures in service}
\label{sec:assessment-in-service}

\subsection{Structural assessment of existing tunnels}
\label{sec:existing-assessment}
\scopenote{Assessing masonry, cast iron and early concrete linings with
unknown properties and geometry; back-analysis from observed condition;
assessment against modern loading.}

\subsection{Instrumentation of operating assets}
\label{sec:operational-monitoring}
\scopenote{Long-term monitoring of movement and water ingress, and the use of
third-party works monitoring when new construction passes nearby
(Chapter~\ref{ch:settlement-damage}).}

\section{Repair, strengthening and upgrading}
\label{sec:repair-strengthening-and-upgrading}

\subsection{Repair and remediation}
\label{sec:repair}
\scopenote{Crack injection, patch repair, leak sealing and grouting, drainage
rehabilitation, and cathodic protection; selecting a repair that treats the
cause rather than the symptom.}

\subsection{Strengthening and relining}
\label{sec:relining}
\scopenote{Sprayed concrete relining, segmental and sleeve relining,
structural linings and the loss of clearance they impose.}

\subsection{Upgrading heritage and legacy tunnels}
\label{sec:heritage}
\scopenote{Nineteenth and twentieth century tunnels brought to modern
clearance, fire and safety standards, and the conflict with heritage
constraints.}

\section{Operational safety and whole-life management}
\label{sec:operational-safety-and-whole-life-management}

\subsection{Operational safety systems}
\label{sec:operational-safety}
\scopenote{Ventilation and smoke control, fire detection and suppression,
egress and cross-passage spacing, and incident management.}

\subsection{Whole-life cost and asset management systems}
\label{sec:whole-life-cost}
\scopenote{Degradation modelling, intervention optimisation, ISO 55000
frameworks, and the persistent under-funding of maintenance relative to
capital works.}

\section{Decommissioning and repurposing}
\label{sec:decommissioning}
\scopenote{End-of-life options: abandonment and infilling, monitoring in
perpetuity, or repurposing for storage, data centres, energy or transport.}
